# ══════════════════════════════════════════════════════════════════
#  Mini Git — 기능 시연 스크립트 (기능 동작)
#     실행:  python3 main.py --fixed-clock < demo/features.txt
#     --fixed-clock: 09:00 부터 커밋마다 15분 → hash 가 매번 같다
#     기대 출력:  demo/features.out  (diff 로 비교 가능)
#  최종 그래프 (화살표 = 자식 → 부모):
#     A ← P ← M1 ← M2 ← S    A  4280179 Set up project       P  7333cf5 Add cart
#     A ← L ← F  ← M2        L  1cdd61f Add signup            F  8ff2dea Fix signup bug
#         L ← M1             M1 70cfbfe merge(P, L)          M2 a16e116 merge(M1, F)
#                            S  828a093 시계가 틀린 커밋 (08:00)
# ══════════════════════════════════════════════════════════════════

# ── 0. INIT 전에는 아무것도 못 한다 ──────────────────────────────
commit "too early"

# ── 1-1. INIT: main 브랜치 · HEAD · 현재 사용자 ────────────────────
init "Ari"
# * = HEAD. 아직 커밋이 없어서 main 은 아무것도 안 가리킨다
branch

# ── 1-2. BRANCH → SWITCH → COMMIT 이 그 브랜치에 반영되는가 ────────
commit "Set up project"
branch feature
# BRANCH 는 이름표만 붙인다: main · feature 가 같은 커밋, HEAD 는 아직 main
branch
switch feature
commit "Add signup feature"
# feature 만 전진했다 (main 은 Set up project 그대로)
branch
switch main
commit "Add cart feature"
branch

# ── 1-3. LOG: 부모가 항상 자식보다 먼저 ─────────────────────────
merge feature
init "Ben"
switch feature
commit "Fix SIGNUP, signout bug"
switch main
merge feature
# merge 커밋은 두 부모가 모두 나온 뒤에야 나온다 (Merge: 줄 = 부모 2개)
log

# 시계가 틀린 컴퓨터: 자식인데 부모보다 '과거' 시각
commit --date="2025-03-10 08:00:00" "Commit from a skewed clock"
# LOG 는 시각이 아니라 관계로 정렬 → 그래도 맨 마지막(부모 다음)
log

# ── 1-4. PATH: 최단 경로 / 동률이면 사전순 / 없으면 No path ────────
# 같은 커밋 → 0칸
path HEAD HEAD
# 부모-자식 → 1칸 (시계 틀린 커밋 → M2)
path 828a093 a16e116
# 형제 브랜치 F → P: 간선을 무방향으로 보므로 공통 조상 A 로 '올라갔다 내려온다'
# 3칸짜리 후보가 F-L-A-P · F-L-M1-P · F-M2-M1-P 셋 → 문자열이 가장 작은 것
path 8ff2dea 7333cf5
# 동률 2칸: P → L 은 P-A-L(4280179) 과 P-M1-L(70cfbfe) → '4' < '7' 이라서 A 쪽
path 7333cf5 1cdd61f
# 새 루트를 만들면 기존 이력과 연결이 끊긴다
switch --orphan docs
commit "Docs root"
path docs main
switch main

# ── 1-5. ANCESTORS: 모든 조상, 빠짐없이 · 중복 없이 ────────────────
# L 은 M1 경유 · F 경유 두 갈래로 닿지만 한 번만 나온다
ancestors HEAD
# 루트의 조상은 0개
ancestors 4280179

# ── 1-6. SEARCH (역색인) ─────────────────────────────────────────
# 대소문자 무시 + 'SIGNUP,' 의 쉼표도 떼고 색인했다
search signup
# 여러 단어 = 모두 포함 (AND)
search "fix bug"
search nothing
search --author=ben
search --author=ARI

# ── 1-7. LOG --sort-by=date | author ────────────────────────────
# date: 시계가 틀린 커밋이 맨 앞으로 (시각 기준)
log --sort-by=date
# author: Ari 묶음 → Ben 묶음, 같은 작성자 안에서는 이력 순서 유지 (안정 정렬)
log --sort-by=author

# ── 1-8. 표준 에러 메시지 ─────────────────────────────────────────
switch nope
ancestors zzzzzzz
path HEAD
log --sort-by=size
search --author
commit
frobnicate
commit "unclosed
exit
